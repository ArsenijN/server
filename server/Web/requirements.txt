# FluxDrop server dependencies
# Tested on Python 3.14.3 (minimum recommended version — server is developed and
# run on Python 3.14.3; older 3.x releases may work but are not supported).
#
# Install:
#   pip install -r requirements.txt
# or into a venv (recommended for production):
#   python3.14 -m venv .venv && source .venv/bin/activate
#   pip install -r requirements.txt

# --- Multipart form parsing ---
# Replaces the deprecated cgi module removed in Python 3.13+.
# parse_form_data() is used in server_cdn.py for file upload parsing.
werkzeug>=3.0.0

# --- Image processing ---
# Used for avatar upload processing (resize + AVIF/WebP/JPEG conversion).
# AVIF output requires Pillow ≥ 9.1 AND libavif on the system:
#   Ubuntu/Debian: sudo apt install libavif-dev
# Without libavif, Pillow falls back to WebP then JPEG automatically.
#   pip install Pillow
Pillow>=10.0.0

# --- Password hashing ---
bcrypt>=4.1.0

# --- Dev / testing (not required for production) ---
# pytest>=8.0.0
# black>=24.0.0

# --- Required for TLS checks in health monitoring (optional, but recommended) ---
cryptography>=41.0.0
